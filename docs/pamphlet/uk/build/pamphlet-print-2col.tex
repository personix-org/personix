% NOT a committed deliverable yet — experimental. Same "safe frame" print
% geometry as pamphlet-print.tex (bleed, shrunk images, inward page numbers,
% font ×1.05).
%
% --- TUNABLE CONSTANTS ----------------------------------------------------
%   Column gap   : \columnsep (space between the two text columns)
%   Bleed        : 3 mm  (media 216×154 = A5 landscape trim + 3 mm/side)
%   Image frame  : \figframelen (cream margin of full-page images, from TRIM)
%   Font size    : Palatino Scale=1.05
% -------------------------------------------------------------------------

% Tolerate looser column balancing — multicol's default flush balancing on a
% \raggedbottom page otherwise triggers a flood of harmless "Overfull \vbox
% while \output is active" warnings.
\setlength{\multicolsep}{12pt plus 4pt minus 3pt}
% If fewer than ~6 lines remain after a full-width break (callout/table/figure),
% multicol DEFERS the next column block to a fresh page instead of starting it
% with a near-empty left column / orphaned heading in the right one. This is
% Pavel's "push the start to the next page" rule, enforced automatically.
\setlength{\premulticols}{6\baselineskip}
\setlength{\postmulticols}{0pt}
% Fill columns left-to-right rather than balancing to equal height. Balancing
% was leaving the left column empty and dumping a short post-callout section
% into the right column; ragged fill keeps the natural left-first flow.
\raggedcolumns

% callout that does not fit at the bottom of the page moves whole to the next

  linewidth=0.7pt,
  roundcorner=5pt,

}

\newcommand{\cbsplitfirst}[1]{%
  \draw[line width=0.7pt, color=#1!50] (O) -- (P|-O);
  \node[anchor=south east, font=\itshape\small,
        text=#1!75, inner xsep=14pt, inner ysep=4pt]
    at (P|-O) {continues overleaf};%
}
\newcommand{\cbsplitsecond}[1]{%
  \draw[line width=0.7pt, color=#1!50] (P-|O) -- (P);
  \node[anchor=north east, font=\itshape\small,
        text=#1!75, inner xsep=14pt, inner ysep=4pt]
    at (P) {continued from overleaf};%
}

  \vspace{2pt}%
  \noindent{\color{#1!40}\leavevmode\leaders\hrule height 0.4pt\hfill\kern0pt}\par
  \vspace{4pt}%
}

% --- Full-page figure helper (PRINT: safe frame) ---
\usepackage{afterpage}
\newcommand{\fullpagefigureNOW}[1]{%
  \clearpage
  \thispagestyle{empty}%
  \begin{tikzpicture}[remember picture, overlay]
    \node[anchor=center, inner sep=0pt] at (current page.center)
      {\includegraphics[width=\dimexpr\paperwidth-2\bleedlen-2\figframelen\relax,keepaspectratio]{#1}};
  \end{tikzpicture}%
  \mbox{}%
  \clearpage
}
% Now that callouts are atomic (never split), the figure can again be deferred
% via \afterpage so the in-progress text page fills to its natural end before
% the figure takes the next page — fewer half-empty text pages before figures.
% Consecutive figure runs still use the immediate variant (master wires them).
\newcommand{\fullpagefigure}[1]{\afterpage{\fullpagefigureNOW{#1}}}
\newcommand{\fullpagefigureimmediate}[1]{\fullpagefigureNOW{#1}}
