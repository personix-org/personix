% This is a SEPARATE copy of pamphlet.tex on purpose: the screen master stays
% untouched. The *content* is shared — both masters \input the same generated
% chapters/ and frontmatter/backmatter — so editing the markdown propagates to
% both. Only the PRINT-SPECIFIC preamble differs (this file's top block).
%
% If you ever change the chapter structure or preamble in pamphlet.tex, port
% the relevant change here too. Built via: ./build.sh --print
%
% Printer strategy chosen: "safe frame" (varianta B) — nothing runs to the cut
% edge. Full-page images are shrunk and centred on the cream background, so an
% imperfect trim always cuts into cream, never into the image. Page numbers are
% pulled inward. A 3 mm bleed is still added to the media box (cream fills it),
% so the PDF is also safe if the printer's workflow expects bleed.
%
% --- TUNABLE PRINT CONSTANTS (adjust to the printer's spec) ---------------
%   Bleed        : 3 mm  → media box = A5 landscape trim (210×148) + 3 mm/side
%                          = 216 × 154 mm. Set in \geometry below + \bleedlen.
%   Image frame  : \figframelen (cream margin of full-page images, from TRIM)
%   Page-number  : footer offset 3 mm (screen uses 8 mm → pulled 5 mm inward)
%   Font size    : Palatino Scale=1.05 (≈ 11.5 pt — "a touch larger" per printer)
% -------------------------------------------------------------------------
% openright: chapters start on a recto (right-hand) page, as a book is normally
% typeset. LaTeX auto-inserts a blank verso (vakát) before a chapter that would
% otherwise land on a verso — counted into the page total below.

% --- Full-page figure helper (PRINT: safe frame, not full-bleed) ---
% Image is shrunk to leave a cream frame of \figframelen from the TRIM edge on
% the long side; aspect ratio preserved and centred on the page, so the trim
% always cuts cream. The safe-frame page colour is tinted to THIS figure's own
% background (figbg@<basename>, from figbg.tex) so the frame blends with the
% image instead of clashing with the cooler global papercream; restored to
% papercream afterwards for the text pages. Page numbers omitted on figure
% pages (graphics carry their own banner text); the counter still advances.
% PRINT (spadačka + soft edge): full-page graphic keeps its ratio, is shrunk to
% leave a ≥5 mm margin to the trim on every side, centred. The margin (and the
% 3 mm bleed) is tinted to that figure's CONTACT-EDGE colour (figbg@<basename>),
% and the figure's own edge is feathered into that tint (figures-bleed/ version),
% so the join blends with no seam. Content stays crisp; papercream restored after.
\usepackage{afterpage}
\newcommand{\fullpagefigureNOW}[1]{%
  \clearpage
  \thispagestyle{empty}%
  \StrBehind{#1}{/}[\fpffile]%
  \StrBefore{\fpffile}{.png}[\fpfbase]%
  \ifcsname figbgset@\fpfbase\endcsname
    \edef\fpfcol{figbg@\fpfbase}%
    \expandafter\pagecolor\expandafter{\fpfcol}%
  \else
    \pagecolor{papercream}%
  \fi
  \begin{tikzpicture}[remember picture, overlay]
    \node[anchor=center, inner sep=0pt] at (current page.center)
      {\IfFileExists{figures-bleed/\fpffile}%
        {\includegraphics[width=\dimexpr\paperwidth-2\bleedlen-2\figframelen\relax,height=\dimexpr\paperheight-2\bleedlen-2\figframelen\relax,keepaspectratio]{figures-bleed/\fpffile}}%
        {\includegraphics[width=\dimexpr\paperwidth-2\bleedlen-2\figframelen\relax,height=\dimexpr\paperheight-2\bleedlen-2\figframelen\relax,keepaspectratio]{#1}}};
  \end{tikzpicture}%
  \mbox{}%
  \clearpage
  \pagecolor{papercream}% restore for following text pages
}
\newcommand{\fullpagefigure}[1]{\afterpage{\fullpagefigureNOW{#1}}}
\newcommand{\fullpagefigureimmediate}[1]{\fullpagefigureNOW{#1}}
