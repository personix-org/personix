% v6 (revised): default callouts are ATOMIC again — splitting a 4–5 line
% box can leave a title-only first half (Pavel's feedback re: Voluntariness
% spread). Splitting is opt-in via the cbXxxsplitbox environments (used for
% long callouts like Incorruptibility), where the title + multiple body
% lines always fit on the first half.
% needspace=9 baselines ≈ title row + 5 body lines + padding — covers
% typical short callouts; if the bottom of the page has less, mdframed
% defers the whole box. Combined with \raggedbottom this no longer over-
% stretches the preceding page, but it does leave some bottoms shorter
% than the figure-and-text rhythm — an acceptable trade-off when the
% alternative is split-with-title-only.
\mdfdefinestyle{calloutbase}{%
  linewidth=0.7pt,
  roundcorner=5pt,
  innerleftmargin=14pt,innerrightmargin=14pt,
  innertopmargin=10pt,innerbottommargin=11pt,
  skipabove=10pt,skipbelow=10pt,
  needspace=9\baselineskip,
  nobreak=true,
  splittopskip=32pt,splitbottomskip=54pt,
}
% v6: toggle helpers retained for backwards compat — \cbatomic forces a single
% box to refuse breaking (use sparingly for very short emphatic callouts).
\newcommand{\cballowsplit}{\mdfsetup{nobreak=false}}
\newcommand{\cbatomic}{\mdfsetup{nobreak=true}}
% Subdued "continues / continued" overlays + closing borders so each split
% half looks like a fully framed box. mdframed defaults leave the seam open
% (no bottomline on first half, no topline on second); we draw thin matching
% lines manually.
% In firstextra/secondextra: O = bottom-left, P = top-right of THAT half.
\newcommand{\cbsplitfirst}[1]{%
  % close the bottom edge of the top half
  \draw[line width=0.7pt, color=#1!50] (O) -- (P|-O);
  % v6: neutral "overleaf" marker — covers the case where the next page is a
  % full-page figure and the callout actually resumes two pages later.
  \node[anchor=south east, font=\itshape\small,
        text=#1!75, inner xsep=14pt, inner ysep=4pt]
    at (P|-O) {continues overleaf};%
}
\newcommand{\cbsplitsecond}[1]{%
  % close the top edge of the bottom half
  \draw[line width=0.7pt, color=#1!50] (P-|O) -- (P);
  \node[anchor=north east, font=\itshape\small,
        text=#1!75, inner xsep=14pt, inner ysep=4pt]
    at (P) {continued from overleaf};%
}
% Title row: sans-serif, bold, in box colour, with a small dot leader.
% Whole title (rule + glyph + text) lives inside a raggedright minipage so
% long titles wrap cleanly. The horizontal divider below uses a leaders-based
% \hrulefill rather than \rule{\linewidth} — the latter triggered a 17pt
% overfull \hbox warning on every callout because of color-stack overhead.
\newcommand{\cbtitle}[3]{%
  \noindent
  \begin{minipage}{\linewidth}\raggedright%
    \textcolor{#1}{\rule[-0.3ex]{2.6ex}{2.6ex}}\hspace{0.6em}%
    {\sffamily\bfseries\color{#1}\large #3}%
  \end{minipage}\par
  \vspace{2pt}%
  \noindent{\color{#1!40}\leavevmode\leaders\hrule height 0.4pt\hfill\kern0pt}\par
  \vspace{4pt}%
}

% --- Full-page figure helper ---
% v6: graphic is deferred via \afterpage so the *current* text page is allowed
% to fill before the figure consumes the next page. Without this trick,
% \clearpage right after the paragraph end forced an early break — a 2- or
% 3-line tail of the paragraph would be marooned on the page above the figure
% with 80 % blank below (v5 pages 38, 40, 69, 112 …).
%
% Each graphic is full-bleed (the manual GIMP-filled WebP already has A5
% landscape aspect ratio, so no letterbox). Page numbers are intentionally
% omitted on figure pages — most graphics carry banner text along the bottom
% edge that would clash with the numeral. The page counter still advances, so
% continuous numbering is preserved in the surrounding text pages.
\usepackage{afterpage}
% v6: shared low-level macro — drops the figure on the current page.
% (EN build: bez DRAFT overlay. V CZ buildu je tikz overlay zachovaný.)
\newcommand{\fullpagefigureNOW}[1]{%
  \clearpage
  \thispagestyle{empty}%
  \begin{tikzpicture}[remember picture, overlay]
    \node[anchor=center, inner sep=0pt] at (current page.center)
      {\includegraphics[width=\paperwidth,height=\paperheight]{#1}};
  \end{tikzpicture}%
  \mbox{}%
  \clearpage
}
% v6 default: defer via \afterpage so the in-progress text page may fill
% to its natural end before the figure consumes the next page.
\newcommand{\fullpagefigure}[1]{\afterpage{\fullpagefigureNOW{#1}}}
% v6 escape hatch: immediate placement, no afterpage. Use this for runs of
% consecutive figures (e.g. the DID drill-down block) where afterpage queue
% interactions otherwise inject blank pages between them.
\newcommand{\fullpagefigureimmediate}[1]{\fullpagefigureNOW{#1}}

% v6: switch from \flushbottom (book class default) to \raggedbottom.
% Rationale: with the previous nobreak=true callouts, \flushbottom couldn't
% absorb the resulting half-page voids and the pages still looked half-empty,
% just with even spacing. Combined with splittable callouts + widow penalties,
% \raggedbottom now produces text that fills naturally to where the next
% break is sensible, and the slight unevenness of the bottom margin is the
% classical book-typography signature of a well-set page.
\raggedbottom
