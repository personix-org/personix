% --- Full-page figure helper ---
% v6: graphic is deferred via \afterpage so the *current* text page is allowed
% to fill before the figure consumes the next page.
% PRINT (spadačka + měkký okraj): celostránková grafika si zachová poměr stran,
% je zmenšená tak, aby od TRIM hrany zbyla mezera \figframelen (5 mm) z každé
% strany, a vystředěná. Mezera (i 3mm spadávka za řez) je vyplněná barvou MÍSTA
% STYKU té grafiky (figbg@<basename> z figbg.tex) — barvou jejího vlastního
% pozadí přímo u okraje. Navíc se okraj grafiky FEATHERem rozpustí do té barvy
% (verze z figures-bleed/), takže přechod splyne a nevznikne šev. Obsah zůstává
% ostrý; po grafice se vrátí papercream pro textové strany.
\usepackage{afterpage}
\newcommand{\fullpagefigureNOW}[1]{%
  \clearpage
  \thispagestyle{empty}%
  \StrBehind{#1}{/}[\fpffile]%
  \StrBefore{\fpffile}{.png}[\fpfbase]%
  \ifcsname figbgset@\fpfbase\endcsname
    \edef\fpfcol{figbg@\fpfbase}%
    \expandafter\pagecolor\expandafter{\fpfcol}%
  \else
    \pagecolor{papercream}%
  \fi
  \begin{tikzpicture}[remember picture, overlay]
    \node[anchor=center, inner sep=0pt] at (current page.center)
      {\IfFileExists{figures-bleed/\fpffile}%
        {\includegraphics[width=\dimexpr\paperwidth-2\bleedlen-2\figframelen\relax,height=\dimexpr\paperheight-2\bleedlen-2\figframelen\relax,keepaspectratio]{figures-bleed/\fpffile}}%
        {\includegraphics[width=\dimexpr\paperwidth-2\bleedlen-2\figframelen\relax,height=\dimexpr\paperheight-2\bleedlen-2\figframelen\relax,keepaspectratio]{#1}}};
  \end{tikzpicture}%
  \mbox{}%
  \clearpage
  \pagecolor{papercream}% obnova pro následující textové strany
}
% v6 default: defer via \afterpage so the in-progress text page may fill
% to its natural end before the figure consumes the next page.
\newcommand{\fullpagefigure}[1]{\afterpage{\fullpagefigureNOW{#1}}}

\newcommand{\fullpagefigureimmediate}[1]{\fullpagefigureNOW{#1}}
